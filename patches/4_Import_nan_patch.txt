
# PATCH for pages/4_Import.py to normalize Mint Mark & Variety
# 1) Add near the top:
NAN_LIKE = {"nan","NaN","none","None","-","—"}
def norm_text(x):
    if x is None: return ""
    s = str(x).strip()
    return "" if s in NAN_LIKE or s.lower() in NAN_LIKE else s

# 2) Wherever you read the mapped columns for mint_mark / variety before calling upsert_coin_type, do:
mint_mark = norm_text(mapped_row.get('mint_mark'))
variety   = norm_text(mapped_row.get('variety'))
coin_type_id = upsert_coin_type(master_id, int(year), mint_mark, variety)
